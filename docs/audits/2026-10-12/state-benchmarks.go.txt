package state

import (
 "encoding/json"
 "fmt"
 "strings"
 "testing"

 "github.com/spk/spk-mm-client/internal/mm/model"
)

func auditLongHistory(n int)*Server{
 s:=newFixture();s.active="town"
 posts:=make([]model.Post,n)
 for i:=range posts { posts[i]=model.Post{ID:fmt.Sprintf("post-%d",i),ChannelID:"town",UserID:"u2",Message:strings.Repeat("text ",100),CreateAt:int64(i+1)} }
 s.older=posts[:n-60];s.chans["town"].Win=Window{Loaded:true,Posts:posts[n-60:]}
 return s
}

func TestAuditLongHistoryWireSize(t *testing.T){
 for _,n:=range []int{60,1000,10000,50000}{
  s:=auditLongHistory(n);v,_:=s.ChannelView("town");wire,err:=json.Marshal(v)
  if err!=nil{t.Fatal(err)};t.Logf("AUDIT posts=%d JSON bytes=%d",n,len(wire))
 }
}

func BenchmarkAuditChannelView(b *testing.B){
 for _,n:=range []int{60,1000,10000,50000}{
  b.Run(fmt.Sprint(n),func(b *testing.B){
   s:=auditLongHistory(n);b.ReportAllocs();b.ResetTimer()
   for b.Loop(){v,_:=s.ChannelView("town");if len(v.Posts)!=n{b.Fatal("posts")}}
  })
 }
}

func BenchmarkAuditChannelViewAndJSON(b *testing.B){
 for _,n:=range []int{60,1000,10000,50000}{
  b.Run(fmt.Sprint(n),func(b *testing.B){
   s:=auditLongHistory(n);b.ReportAllocs();b.ResetTimer()
   for b.Loop(){v,_:=s.ChannelView("town");if _,e:=json.Marshal(v);e!=nil{b.Fatal(e)}}
  })
 }
}
